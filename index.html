// ===== 時計目盛りプラグイン =====

const clockTicks = {

id: 'clockTicks',

afterDraw(chart) {

const {ctx} = chart;

const cx = chart.width / 2;

const cy = chart.height / 2;

const r = chart._metasets[0].data[0].outerRadius + 10;


ctx.save();

ctx.textAlign = "center";

ctx.textBaseline = "middle";

ctx.font = "10px sans-serif";


for(let i=0;i<24;i++){

let angle = (i/24)*2*Math.PI - Math.PI/2;


let x = cx + Math.cos(angle)*r;

let y = cy + Math.sin(angle)*r;


ctx.fillText(i, x, y);


let x1 = cx + Math.cos(angle)*(r-8);

let y1 = cy + Math.sin(angle)*(r-8);


ctx.beginPath();

ctx.moveTo(x1,y1);

ctx.lineTo(x,y);

ctx.strokeStyle="#aaa";

ctx.stroke();

}

ctx.restore();

}

};


// ===== 時間処理 =====

function t2m(t){

let [h,m]=t.split(":");

return h*60 + +m;

}


function span(s,e){

if(e<s) e+=1440;

return e-s;

}


function overlap(a,b,c,d){

if(b<a)b+=1440;

if(d<c)d+=1440;

return Math.max(a,c) < Math.min(b,d);

}


// ===== 時計 =====

let chart;


function updateClock(){


let s1=t2m(sleepStart.value);

let s2=t2m(sleepEnd.value);

let sc1=t2m(schoolStart.value);

let sc2=t2m(schoolEnd.value);


let sleep=span(s1,s2);

let school=span(sc1,sc2);

let free=1440 - sleep - school;


let over=overlap(s1,s2,sc1,sc2);


warn.innerText = over ? "⚠️ 時間かぶってる！" : "";


if(chart) chart.destroy();


chart = new Chart(clockChart,{

type:"doughnut",

data:{

datasets:[{

data:[sleep,school,free],

backgroundColor:[

over ? "#ff6b6b" : "#caffbf",

"#a0c4ff",

"#ffd6a5"

]

}]

},

options:{

rotation:-90,

circumference:360,

plugins:{legend:{display:false}}

},

plugins:[clockTicks]

});

}


// ===== ToDo =====

let todo = JSON.parse(localStorage.getItem("todo")||"[]");


function renderTodo(){

todoList.innerHTML = todo.map((t,i)=>`

<div class="todo-item">

<label>

<input type="checkbox" ${t.done?"checked":""} onclick="toggleTodo(${i})">

${t.text}

</label>

<div>

<button onclick="addCal('${t.text}')">📅</button>

<button onclick="delTodo(${i})">×</button>

</div>

</div>`).join("");


localStorage.setItem("todo",JSON.stringify(todo));

}


function addTodo(){

let v=todoInput.value;

if(!v)return;

todo.push({text:v,done:false});

todoInput.value="";

renderTodo();

}


function toggleTodo(i){

todo[i].done=!todo[i].done;

renderTodo();

}


function delTodo(i){

todo.splice(i,1);

renderTodo();

}


// ===== カレンダー =====

function addCal(text){

let t=encodeURIComponent(text);

let d=new Date();

let y=d.getFullYear();

let m=String(d.getMonth()+1).padStart(2,'0');

let da=String(d.getDate()).padStart(2,'0');


let url=`https://calendar.google.com/calendar/render?action=TEMPLATE&text=${t}&dates=${y}${m}${da}/${y}${m}${da}`;


window.open(url);

}


// ===== 持ち物 =====

let items = JSON.parse(localStorage.getItem("items")||"[]");


function renderItems(){

itemList.innerHTML = items.map((t,i)=>`

<div class="todo-item">

<label><input type="checkbox"> ${t}</label>

<button onclick="delItem(${i})">×</button>

</div>`).join("");


localStorage.setItem("items",JSON.stringify(items));

}


function addItem(){

let v=itemInput.value;

if(!v)return;

items.push(v);

itemInput.value="";

renderItems();

}


function delItem(i){

items.splice(i,1);

renderItems();

}


// ===== 初期表示 =====

updateClock();

renderTodo();

renderItems();
